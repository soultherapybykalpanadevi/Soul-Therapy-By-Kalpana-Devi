-- Allow the authenticated admin to read bookings. GRANT SELECT ON TABLE public.bookings TO authenticated; CREATE POLICY "Admin can view bookings" ON public.bookings FOR SELECT TO authenticated USING ( (SELECT auth.jwt() ->> 'email') = 'YOUR_ADMIN_EMAIL' ); -- Allow the authenticated admin to update bookings. GRANT UPDATE ON TABLE public.bookings TO authenticated; CREATE POLICY "Admin can update bookings" ON public.bookings FOR UPDATE TO authenticated USING ( (SELECT auth.jwt() ->> 'email') = 'YOUR_ADMIN_EMAIL' ) WITH CHECK ( (SELECT auth.jwt() ->> 'email') = 'YOUR_ADMIN_EMAIL' );
